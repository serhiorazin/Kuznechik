// Function: FN_CHECK_REQUEST_STATUS
// Returns a String[]:
//   result[0] = response code as string
//   result[1] = response body as string (for logging)
// Uses process variables:
//   #CN_URL_IABS
//   #IN_GET_CUSTOMER_REQUESTID
//   #IN_GET_CUSTOMER_TOKEN

String baseUrl = #CN_URL_IABS;
String requestId = "create_cust_" + #IN_GET_CUSTOMER_REQUESTID;
String token = #IN_GET_CUSTOMER_TOKEN;
String[] result = new String[2];
result[0] = "-1";
result[1] = "";

javax.net.ssl.HttpsURLConnection connection = null;
try {
    URL url = new URL(baseUrl + "/1.0.0/check-request/" + requestId);
    connection = (javax.net.ssl.HttpsURLConnection) url.openConnection();
    connection.setConnectTimeout(10000);
    connection.setReadTimeout(90000);
    connection.setRequestMethod("GET");
    connection.setRequestProperty("accept", "application/json");
    connection.setRequestProperty("requestId", requestId);
    connection.setRequestProperty("Accept-Language", "en");
    connection.setRequestProperty("Authorization", "Bearer " + token);

    int responseCode = connection.getResponseCode();

    java.io.InputStream stream;
    if (responseCode >= 400) {
        stream = connection.getErrorStream();
    } else {
        stream = connection.getInputStream();
    }

    StringBuilder res = new StringBuilder();
    if (stream != null) {
        java.io.BufferedReader in = new java.io.BufferedReader(
                new java.io.InputStreamReader(stream, "utf-8")
        );
        String inputLine;
        while ((inputLine = in.readLine()) != null) {
            res.append(inputLine);
        }
        in.close();
    }

    result[0] = String.valueOf(responseCode);
    result[1] = res.toString();
} catch (Exception e) {
    result[0] = "-1";
    result[1] = "Error: " + e.getMessage();
} finally {
    if (connection != null) {
        connection.disconnect();
    }
}

return result;
